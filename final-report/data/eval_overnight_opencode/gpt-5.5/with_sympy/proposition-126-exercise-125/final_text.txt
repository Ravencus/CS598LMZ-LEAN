theorem subsequence_has_same_limit
    {α : Type*} [TopologicalSpace α] {u : ℕ → α} {l : α}
    (hu : Filter.Tendsto u Filter.atTop (nhds l)) :
    ∀ φ : ℕ → ℕ, StrictMono φ →
